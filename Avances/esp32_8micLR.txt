// ─────────────────────────────────────────────────────────────────────────────
// esp32_8micLR.txt  (renombra a .ino antes de abrir en Arduino IDE)
//
// Sistema completo: dos ESP32-S3 + SYNC. Reconstruido en sept 2026 después de
// encontrar la causa raíz de un mes de fallas de sincronización: Serial y
// Serial0 comparten el mismo periférico UART físico en este hardware, así
// que deben abrirse EXACTAMENTE al mismo baud que usa Python (3,000,000) —
// antes se asumía (incorrectamente) que el baud no importaba para USB CDC.
// Ver Documentation.md sección 10 para el detalle completo de la reconstrucción
// por etapas (los archivos de prueba de cada etapa quedaron en
// Exploration/reconstruccion_sept2026/).
//
// El loop() quedó como lectura+escritura directa (sin tareas FreeRTOS
// separadas) porque, con el baud correcto, un loop simple ya sostiene los
// 4 mics por placa sin underruns.
//
// Flashear dos veces cambiando IS_MASTER:
//   IS_MASTER 1 → ESP32-S3 #1 (Master, azul,  genera SYNC, COM del primer ESP)
//   IS_MASTER 0 → ESP32-S3 #2 (Slave,  rojo,   espera SYNC, COM del segundo ESP)
//
// Protocolo de canal — bits [1:0] de cada muestra:
//   00 → Mic 1 (Bus0-L)   01 → Mic 2 (Bus0-R)
//   10 → Mic 3 (Bus1-L)   11 → Mic 4 (Bus1-R)
//   (Python distingue ESP1/ESP2 por el puerto COM)
//
// Conexiones (idénticas en ambas placas):
//   Bus I2S 0  →  WS=4   SCK=5   SD=6    Mic1(L/R=GND) + Mic2(L/R=VCC)
//   Bus I2S 1  →  WS=11  SCK=12  SD=13   Mic3(L/R=GND) + Mic4(L/R=VCC)
//   SYNC       →  GPIO 10 (salida en Master, entrada en Slave — unir ambas
//                 placas por este pin Y compartir GND entre ambas)
//   LED RGB    →  GPIO 38 (NeoPixel WS2812)
//
// Configuración Arduino IDE:
//   esp32 by Espressif → 3.3.11 o superior
//   Board           →  ESP32S3 Dev Module
//   USB CDC On Boot →  Enabled
//   USB Mode        →  Hardware CDC and JTAG
//   Flash Size      →  16MB (128Mb)
//   PSRAM           →  OPI PSRAM
// ─────────────────────────────────────────────────────────────────────────────

#include <Arduino.h>
#include <driver/i2s.h>
#include <Adafruit_NeoPixel.h>

// ── ROL DE ESTA PLACA ─────────────────────────────────────────────────────────
#define IS_MASTER  0    // 1 = Master (azul) | 0 = Slave (rojo)

// ── BAUD (firmware y Python DEBEN coincidir exactamente — ver Etapa 1a) ──────
#define PROJECT_BAUD  3000000

// ── DEBUG ─────────────────────────────────────────────────────────────────────
#define DEBUG true

#if DEBUG
  #define DBG(msg)         Serial0.println(msg)
  #define DBG_FMT(fmt,...) Serial0.printf(fmt, ##__VA_ARGS__)
#else
  #define DBG(msg)
  #define DBG_FMT(fmt,...)
#endif

// ── LED RGB ───────────────────────────────────────────────────────────────────
#define LED_PIN   38
#define NUM_LEDS  1
Adafruit_NeoPixel led(NUM_LEDS, LED_PIN, NEO_GRB + NEO_KHZ800);

// ── SYNC ──────────────────────────────────────────────────────────────────────
#define SYNC_PIN  10

// ── Pines Bus I2S 0 ───────────────────────────────────────────────────────────
#define I2S0_WS   4
#define I2S0_SCK  5
#define I2S0_SD   6

// ── Pines Bus I2S 1 ───────────────────────────────────────────────────────────
#define I2S1_WS   11
#define I2S1_SCK  12
#define I2S1_SD   13

// ── Config de audio ───────────────────────────────────────────────────────────
#define SAMPLE_RATE   16000
#define CHUNK_SAMPLES 512
#define BUS_SAMPLES   (CHUNK_SAMPLES * 2)

// ── IDs de canal en bits [1:0] ────────────────────────────────────────────────
#define ID_MIC1  0x00
#define ID_MIC2  0x01
#define ID_MIC3  0x02
#define ID_MIC4  0x03
#define ID_MASK  0xFFFC

// ── Buffers ───────────────────────────────────────────────────────────────────
static int32_t bus0_buf[BUS_SAMPLES];
static int32_t bus1_buf[BUS_SAMPLES];
static int16_t out_buf[CHUNK_SAMPLES * 4];

// ── LED helpers (colores en orden R,G,B normal — la librería ya compensa
//    el orden físico NEO_GRB internamente, no hay que intercambiar nada) ──────
static void led_color(uint8_t r, uint8_t g, uint8_t b)
{
  led.setPixelColor(0, led.Color(r, g, b));
  led.show();
}

static void led_off()
{
  led.clear();
  led.show();
}

// ── Inicialización I2S ────────────────────────────────────────────────────────
static void i2s_init(i2s_port_t port, int ws, int sck, int sd)
{
  i2s_config_t cfg = {
    .mode                 = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate          = SAMPLE_RATE,
    .bits_per_sample      = I2S_BITS_PER_SAMPLE_32BIT,
    .channel_format       = I2S_CHANNEL_FMT_RIGHT_LEFT,
    .communication_format = I2S_COMM_FORMAT_STAND_I2S,
    .intr_alloc_flags     = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count        = 16,
    .dma_buf_len          = 512,
    .use_apll             = true,
    .tx_desc_auto_clear   = false,
    .fixed_mclk           = 0
  };
  i2s_pin_config_t pins = {
    .bck_io_num   = sck,
    .ws_io_num    = ws,
    .data_out_num = I2S_PIN_NO_CHANGE,
    .data_in_num  = sd
  };

  esp_err_t err = i2s_driver_install(port, &cfg, 0, NULL);
  DBG_FMT("[I2S] driver_install port=%d: %s\n", port, esp_err_to_name(err));
  err = i2s_set_pin(port, &pins);
  DBG_FMT("[I2S] set_pin port=%d ws=%d sck=%d sd=%d: %s\n",
          port, ws, sck, sd, esp_err_to_name(err));
  i2s_start(port);
  DBG_FMT("[I2S] port=%d iniciado\n", port);
}

void setup()
{
  Serial.setTxBufferSize(16384);
  Serial.begin(PROJECT_BAUD);
  Serial0.begin(PROJECT_BAUD);

  DBG_FMT("\n[SETUP] Rol: %s  baud=%d\n", IS_MASTER ? "MASTER" : "SLAVE", PROJECT_BAUD);

  // ── LED: identificación visual ────────────────────────────────────────────
  led.begin();
  led.setBrightness(40);
  led_off();
  delay(100);

#if IS_MASTER
  DBG("[SETUP] LED → AZUL (Master)");
  led_color(0, 0, 200);   // Azul
#else
  DBG("[SETUP] LED → ROJO (Slave)");
  led_color(255, 0, 0);   // Rojo
#endif

  delay(5000);   // 5 segundos de identificación visual
  led_off();
  DBG("[SETUP] LED apagado");

  // ── SYNC GPIO ─────────────────────────────────────────────────────────────
#if IS_MASTER
  pinMode(SYNC_PIN, OUTPUT);
  digitalWrite(SYNC_PIN, LOW);
  DBG_FMT("[SETUP] SYNC GPIO %d → salida (Master)\n", SYNC_PIN);
#else
  pinMode(SYNC_PIN, INPUT);
  DBG_FMT("[SETUP] SYNC GPIO %d → entrada (Slave)\n", SYNC_PIN);
#endif

  // ── Inicializar I2S ───────────────────────────────────────────────────────
  DBG("[SETUP] Inicializando Bus I2S 0...");
  i2s_init(I2S_NUM_0, I2S0_WS, I2S0_SCK, I2S0_SD);

  DBG("[SETUP] Inicializando Bus I2S 1...");
  i2s_init(I2S_NUM_1, I2S1_WS, I2S1_SCK, I2S1_SD);

  delay(100);

  // ── SYNC: alinear arranque entre Master y Slave ───────────────────────────
#if IS_MASTER
  DBG("[SETUP] Master: esperando 500ms para que Slave esté listo...");
  delay(500);
  DBG("[SETUP] Master: enviando pulso SYNC...");
  digitalWrite(SYNC_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(SYNC_PIN, LOW);
  DBG("[SETUP] Master: pulso SYNC enviado");

  // Parpadeo verde para confirmar SYNC enviado
  led_color(0, 255, 0);
  delay(200);
  led_off();
#else
  DBG("[SETUP] Slave: esperando pulso SYNC del Master...");
  uint32_t t0 = millis();
  bool sync_ok = true;
  while (digitalRead(SYNC_PIN) == LOW) {
    if (millis() - t0 > 15000) {
      DBG("[SETUP] Slave: timeout SYNC — arrancando sin sincronización");
      sync_ok = false;
      break;
    }
  }

  if (sync_ok) {
    DBG_FMT("[SETUP] Slave: SYNC recibido tras %lu ms\n", millis() - t0);
    // Parpadeo verde para confirmar SYNC recibido
    led_color(0, 255, 0);
    delay(200);
    led_off();
  } else {
    // 3 parpadeos rojos: no llegó el pulso SYNC del Master (revisar cableado GPIO10/GND)
    for (int i = 0; i < 3; i++) {
      led_color(255, 0, 0);
      delay(150);
      led_off();
      delay(150);
    }
  }
#endif

  // ── Esperar host USB ──────────────────────────────────────────────────────
  DBG("[SETUP] Esperando host USB...");
  uint32_t t1 = millis();
  while (!Serial) {
    delay(10);
    if (millis() - t1 > 10000) {
      DBG("[SETUP] Timeout host USB — arrancando de todas formas");
      break;
    }
  }
  delay(200);
  DBG("[SETUP] Host conectado. Iniciando stream.");
}

void loop()
{
  size_t bytes0 = 0, bytes1 = 0;

  i2s_read(I2S_NUM_0, bus0_buf, sizeof(bus0_buf), &bytes0, portMAX_DELAY);
  i2s_read(I2S_NUM_1, bus1_buf, sizeof(bus1_buf), &bytes1, portMAX_DELAY);

  uint32_t n = min(bytes0, bytes1) / sizeof(int32_t) / 2;

  for (uint32_t i = 0; i < n; i++) {
    out_buf[i * 4 + 0] = (int16_t)((bus0_buf[i*2+0] >> 16) & ID_MASK) | ID_MIC1;
    out_buf[i * 4 + 1] = (int16_t)((bus0_buf[i*2+1] >> 16) & ID_MASK) | ID_MIC2;
    out_buf[i * 4 + 2] = (int16_t)((bus1_buf[i*2+0] >> 16) & ID_MASK) | ID_MIC3;
    out_buf[i * 4 + 3] = (int16_t)((bus1_buf[i*2+1] >> 16) & ID_MASK) | ID_MIC4;
  }

  static uint32_t frame_dbg = 0;
  if (DEBUG && ++frame_dbg % 500 == 0) {
    DBG_FMT("[LOOP] frame=%lu n=%lu IDs=%d %d %d %d\n",
            frame_dbg, n,
            out_buf[0] & 0x03, out_buf[1] & 0x03,
            out_buf[2] & 0x03, out_buf[3] & 0x03);
  }

  Serial.write((uint8_t*)out_buf, n * 4 * sizeof(int16_t));
}
